mydata <- read.csv("dividendinfo.csv")
# In this dataset we assign a value of 1 to a stock that pays a dividend.
# We assign a value of 0 to a stock that does not pay a dividend. 
# Independent variables are as follows:
# - fcfps: Free cash flow per share (in $)
# - earnings_growth: Earnings growth in the past year (in %)
# - de: Debt to Equity ratio
# - mcap: Market Capitalization of the stock
# - current_ratio: Current Ratio (or Current Assets/Current Liabilities)



#Scale the data
scaleddata<-scale(mydata)

# Normalize the data
normalize <- function(x) {
  return ((x - min(x)) / (max(x) - min(x)))
}
maxmindf <- as.data.frame(lapply(mydata, normalize))

# Training and Test Data
trainset <- maxmindf[1:160, ]
testset <- maxmindf[161:200, ]

softplus <- function(x) log(1+exp(x))

#Build Neural Network
#install.packages("neuralnet")
library(neuralnet)
nn <- neuralnet(dividend ~ ., 
                data=trainset, hidden=c(2,1), linear.output=FALSE, stepmax=2000,
                act.fct=softplus)
nn <- neuralnet(dividend ~ ., 
                data=trainset, hidden=c(2,3,1), linear.output=FALSE, 
                threshold=0.01, algorithm="backprop", learningrate=0.005,
                act.fct=softplus)
nn <- neuralnet(dividend ~ ., 
                data=trainset, hidden=c(2,1), linear.output=FALSE, stepmax=10000,
                act.fct="logistic")
# Results
nn$result.matrix
plot(nn)

#Test the resulting output
nn.results <- compute(nn,  testset[,-1])
results <- data.frame(actual = testset$dividend, prediction = nn.results$net.result)
results
#Round results
roundedresults<-sapply(results,round,digits=0)
roundedresultsdf=data.frame(roundedresults)
# Compute confusion Matrix
confMat<-table(roundedresultsdf$actual,roundedresultsdf$prediction)
confMat
sum(diag(confMat))/sum(confMat)
